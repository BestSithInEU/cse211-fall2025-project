% ── Binary Format (BIN) ──
\subsection*{A.10 Binary Format (BIN) --- Ghost in the Shell Theme}
\addcontentsline{toc}{subsection}{A.10 Binary Format (BIN)}

\begin{lstlisting}[style=plaintext, caption={Binary format specification}]
Header (64 bytes):
  [0-3]   Magic: 0x47535442
  [4-7]   Version: uint32
  [8-15]  Node count: uint64
  [16-23] Edge count: uint64

Node Record (32 bytes):
  [0-7]   id: uint64
  [8-15]  value1: double
  [16-23] value2: double
  [24-27] flags: uint32

Edge Record (24 bytes):
  [0-7]   source: uint64
  [8-15]  target: uint64
  [16-23] weight: double
\end{lstlisting}
